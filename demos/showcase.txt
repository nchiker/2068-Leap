BORDER 0
PAPER 0
INK 7
CLS
PRINT "TS2068 ROM DEMO"
PRINT "STRUCTURED BASIC MACHINE"
BEEP 30,60
GOSUB waitkey
GOTO gfxdemo

gfxdemo:
BORDER 1
CLS
MODE 0
INK 6
FOR I = 0 TO 7
CIRCLE 128,96,I*9
NEXT I
GOSUB waitkey
CLS
INK 2
FOR X = 0 TO 240 STEP 16
LINE X,0 TO 255-X,191
NEXT X
GOSUB waitkey
CLS
GOTO arrdemo

arrdemo:
BORDER 2
INK 4
DIM D(8)
FOR I = 0 TO 7
D(I) = I*I
NEXT I
S = 0
FOR I = 0 TO 7
S = S + D(I)
BLOCK I*30,181-D(I) TO I*30+20,181
NEXT I
PRINT "SUM=" + STR$(S)
PRINT "SIZE=" + STR$(DIMN(D))
GOSUB waitkey
CLS
GOTO sprdemo

sprdemo:
BORDER 3
PRINT "SPRITE DEMO"
INK 5
BLOCK 0,0 TO 7,7
SPRITE GRAB 0,0,0,1,1
CLS
INK 3
CIRCLE 3,180,3
SPRITE GRAB 1,22,0,1,1
CLS
SPRITE SHOW 0,0,0
SPRITE SHOW 1,20,30
X = 0
Y = 0
FOR I = 1 TO 15
X = X + 2
Y = Y + 1
IF X > 31 THEN X = 31
IF Y > 20 THEN Y = 20
SPRITE MOVE 0,Y,X
SPRITE MOVE 1,20-Y,31-X
IF HIT(0,1) <> 0 THEN GOSUB collide
PAUSE 5
NEXT I
SPRITE HIDE 0
SPRITE HIDE 1
GOSUB waitkey
CLS
GOTO snddemo

collide:
BORDER 6
SOUND 7,62
SOUND 8,15
SOUND 1,2
PAUSE 10
SOUND 8,0
BORDER 3
RETURN

waitkey:
IF INKEY$() = "" THEN GOTO waitkey
RETURN

snddemo:
BORDER 5
PRINT "SOUND DEMO"
SOUND 7,62
SOUND 8,15
FOR I = 1 TO 8
SOUND 1,I
PAUSE 8
NEXT I
SOUND 8,0
GOSUB waitkey
CLS
GOTO strdemo

strdemo:
BORDER 7
M$ = "USED: "
N$ = "LABELS SPRITES SND"
PRINT M$ + N$
GOSUB waitkey
CLS
INK 7
PRINT "DEMO COMPLETE"
BORDER 4
STOP
